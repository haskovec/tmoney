      606 +// overlayDropdown places a dropdown string on top of the layout at the given row and column offset.
      607 +func overlayDropdown(layout, dropdown string, colOffset, rowOffset, totalWidth int) string {
      608 +  layoutLines := strings.Split(layout, "\n")
      609 +  dropdownLines := strings.Split(dropdown, "\n")
      610 +
      611 +  for i, dLine := range dropdownLines {
      612 +    targetRow := rowOffset + i
      613 +    if targetRow >= len(layoutLines) {
      614 +      break
      615 +    }
      616 +
      617 +    // Build the new line: prefix + dropdown + suffix
      618 +    bgLine := layoutLines[targetRow]
      619 +    bgRunes := []rune(stripAnsi(bgLine))
      620 +
      621 +    // Build prefix (characters before the dropdown)
      622 +    prefix := ""
      623 +    if colOffset > 0 {
      624 +      if colOffset <= len(bgRunes) {
      625 +        prefix = string(bgRunes[:colOffset])
      626 +      } else {
      627 +        prefix = string(bgRunes) + strings.Repeat(" ", colOffset-len(bgRunes))
      628 +      }
      629 +    }
      630 +
      631 +    // Build suffix (characters after the dropdown)
      632 +    dropdownWidth := lipgloss.Width(dLine)
      633 +    endCol := colOffset + dropdownWidth
      634 +    suffix := ""
      635 +    if endCol < len(bgRunes) {
      636 +      suffix = string(bgRunes[endCol:])
      637 +    }
      638 +
      639 +    _ = totalWidth
      640 +    layoutLines[targetRow] = prefix + dLine + suffix
      641 +  }
      642 +
      643 +  return strings.Join(layoutLines, "\n")
      644 +}
      645 +
      646 +// stripAnsi removes ANSI escape codes from a string for width calculation.
      647 +func stripAnsi(s string) string {
      648 +  var result []rune
      649 +  inEscape := false
      650 +  for _, r := range s {
      651 +    if r == '\033' {
      652 +      inEscape = true
      653 +      continue
      654 +    }
      655 +    if inEscape {
      656 +      if (r >= 'A' && r <= 'Z') || (r >= 'a' && r <= 'z') {
      657 +        inEscape = false
      658 +      }
      659 +      continue
      660 +    }
      661 +    result = append(result, r)
      662 +  }
      663 +  return string(result)
      664 +}
