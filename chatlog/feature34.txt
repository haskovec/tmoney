      148 +func (s *ReportService) spendingByCategory(period string, startDate, endDate time.Time) (*models.SpendingReport, error) {
      149 +  // Query to get spending by category, including both direct transactions and splits.
      150 +  // We need to aggregate:
      151 +  // 1. Direct transactions (non-split) with expense categories
      152 +  // 2. Split transaction amounts with expense categories
      153 +  // Exclude transfers (system categories) and only count negative amounts (spending).
      154 +  query := `
      155 +    WITH category_spending AS (
      156 +      -- Direct transactions (non-split, with category)
      157 +      SELECT
      158 +        c.id AS category_id,
      159 +        c.name AS category_name,
      160 +        c.parent_id,
      161 +        COALESCE(parent.name, '') AS parent_name,
      162 +        ABS(t.amount) AS spending
      163 +      FROM transactions t
      164 +      JOIN categories c ON CAST(t.category_id AS VARCHAR) = CAST(c.id AS VARCHAR)
      165 +      LEFT JOIN categories parent ON CAST(c.parent_id AS VARCHAR) = CAST(parent.id AS VARCHAR)
      166 +      WHERE c.type = 'expense'
      167 +        AND c.system_category = FALSE
      168 +        AND t.date >= ?
      169 +        AND t.date <= ?
      170 +        AND t.amount < 0
      171 +        AND NOT EXISTS (
      172 +          SELECT 1 FROM transaction_splits ts
      173 +          WHERE CAST(ts.transaction_id AS VARCHAR) = CAST(t.id AS VARCHAR)
      174 +        )
      175 +
      176 +      UNION ALL
      177 +
      178 +      -- Split transaction amounts
      179 +      SELECT
      180 +        c.id AS category_id,
      181 +        c.name AS category_name,
      182 +        c.parent_id,
      183 +        COALESCE(parent.name, '') AS parent_name,
      184 +        ABS(ts.amount) AS spending
      185 +      FROM transaction_splits ts
      186 +      JOIN categories c ON CAST(ts.category_id AS VARCHAR) = CAST(c.id AS VARCHAR)
      187 +      LEFT JOIN categories parent ON CAST(c.parent_id AS VARCHAR) = CAST(parent.id AS VARCHAR)
      188 +      JOIN transactions t ON CAST(ts.transaction_id AS VARCHAR) = CAST(t.id AS VARCHAR)
      189 +      WHERE c.type = 'expense'
      190 +        AND c.system_category = FALSE
      191 +        AND t.date >= ?
      192 +        AND t.date <= ?
      193 +        AND ts.amount < 0
      194 +    )
      195 +    SELECT
      196 +      category_id,
      197 +      category_name,
      198 +      parent_id,
      199 +      parent_name,
      200 +      SUM(spending) AS total_spending
      201 +    FROM category_spending
      202 +    GROUP BY category_id, category_name, parent_id, parent_name
      203 +    ORDER BY total_spending DESC
      204 +  `
