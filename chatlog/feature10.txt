       321 +  // MergeCategories merges the source category into the target category.
       322 +  // All transactions, splits, scheduled transactions, and payee defaults
       323 +  // using the source category will be updated to use the target category.
       324 +  // The source category is then deleted.
       325 +  //
       326 +  // Rules:
       327 +  // - Source and target must have the same type (income/expense)
       328 +  // - System categories cannot be merged
       329 +  // - Cannot merge a category into itself
       330 +  func (s *CategoryService) MergeCategories(sourceID, targetID models.ID) error {
       331 +    // Cannot merge into itself
       332 +    if sourceID == targetID {
       333 +      return &CategoryMergeSameError{ID: sourceID.String()}
       334 +    }
       335 +  
       336 +    // Get both categories
       337 +    source, err := s.repo.GetByID(sourceID)
       338 +    if err != nil {
       339 +      return fmt.Errorf("failed to get source category: %w", err)
       340 +    }
       341 +  
       342 +    target, err := s.repo.GetByID(targetID)
       343 +    if err != nil {
       344 +      return fmt.Errorf("failed to get target category: %w", err)
       345 +    }
       346 +  
       347 +    // System categories cannot be merged
       348 +    if source.IsSystem {
       349 +      return &CategoryIsSystemError{ID: sourceID.String(), Name: source.Name}
       350 +    }
       351 +    if target.IsSystem {
       352 +      return &CategoryIsSystemError{ID: targetID.String(), Name: target.Name}
       353 +    }
       354 +  
       355 +    // Types must match
       356 +    if source.Type != target.Type {
       357 +      return &CategoryMergeTypeMismatchError{
       358 +        SourceID:   sourceID.String(),
       359 +        SourceType: source.Type.String(),
       360 +        TargetID:   targetID.String(),
       361 +        TargetType: target.Type.String(),
       362 +      }
       363 +    }
       364 +  
       365 +    // Update all references to use target category
       366 +    // Note: DuckDB doesn't support transactions, so we do best-effort updates
       367 +  
       368 +    // Update transactions
       369 +    _, err = s.db.Conn().Exec(`
       370 +      UPDATE transactions
       371 +      SET category_id = CAST(? AS UUID), updated_at = CURRENT_TIMESTAMP
       372 +      WHERE CAST(category_id AS VARCHAR) = ?
       373 +    `, targetID.String(), sourceID.String())
       374 +    if err != nil {
       375 +      return fmt.Errorf("failed to update transactions: %w", err)
       376 +    }
       377 +  
       378 +    // Update transaction splits
       379 +    _, err = s.db.Conn().Exec(`
       380 +      UPDATE transaction_splits
       381 +      SET category_id = CAST(? AS UUID)
       382 +      WHERE CAST(category_id AS VARCHAR) = ?
       383 +    `, targetID.String(), sourceID.String())
       384 +    if err != nil {
       385 +      return fmt.Errorf("failed to update transaction splits: %w", err)
       386 +    }
       387 +  
       388 +    // Update payee defaults
       389 +    _, err = s.db.Conn().Exec(`
       390 +      UPDATE payees
       391 +      SET default_category_id = CAST(? AS UUID), updated_at = CURRENT_TIMESTAMP
       392 +      WHERE CAST(default_category_id AS VARCHAR) = ?
       393 +    `, targetID.String(), sourceID.String())
       394 +    if err != nil {
       395 +      return fmt.Errorf("failed to update payee defaults: %w", err)
       396 +    }
       397 +  
       398 +    // Update scheduled transactions
       399 +    _, err = s.db.Conn().Exec(`
       400 +      UPDATE scheduled_transactions
       401 +      SET category_id = CAST(? AS UUID), updated_at = CURRENT_TIMESTAMP
       402 +      WHERE CAST(category_id AS VARCHAR) = ?
       403 +    `, targetID.String(), sourceID.String())
       404 +    if err != nil {
       405 +      return fmt.Errorf("failed to update scheduled transactions: %w", err)
       406 +    }
       407 +  
       408 +    // If source has children, reassign them to target
       409 +    children, err := s.repo.ListChildren(sourceID)
       410 +    if err != nil {
       411 +      return fmt.Errorf("failed to list source children: %w", err)
       412 +    }
       413 +    for _, child := range children {
       414 +      child.SetParent(targetID)
       415 +      if err := s.repo.Update(child); err != nil {
       416 +        return fmt.Errorf("failed to reassign child category %s: %w", child.Name, err)
       417 +      }
       418 +    }
       419 +  
       420 +    // Delete the source category (should now have no references)
       421 +    if err := s.repo.Delete(sourceID); err != nil {
       422 +      return fmt.Errorf("failed to delete source category: %w", err)
       423 +    }
       424 +  
       425 +    return nil
       426 +  }
